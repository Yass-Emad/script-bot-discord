def load_json(filename, default=None):
    if default is None:
        default = {}
    if os.path.exists(filename):
        try:
            with open(filename, "r", encoding="utf-8") as f:
                return json.load(f)
        except json.JSONDecodeError as e:
            logger.error(f"ملف {filename} تالف: {e}")
        except Exception as e:
            logger.error(f"فشل تحميل {filename}: {e}")
    return default

def save_json(filename, data):
    try:
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=4, ensure_ascii=False)
    except Exception as e:
        logger.error(f"فشل حفظ {filename}: {e}")

user_cooldowns = {}

def check_cooldown(user_id, seconds=5):
    now = datetime.now(timezone.utc)
    if user_id in user_cooldowns:
        last_used = user_cooldowns[user_id]
        if (now - last_used).total_seconds() < seconds:
            return False, seconds - int((now - last_used).total_seconds())
    user_cooldowns[user_id] = now
    return True, 0

def get_queue(guild_id):
    if guild_id not in queues:
        queues[guild_id] = asyncio.Queue()
    return queues[guild_id]

def is_loop_on(guild_id):
    return loop_states.get(guild_id, False)

def set_loop(guild_id, state: bool):
    loop_states[guild_id] = state

def get_volume(guild_id):
    return volume_levels.get(guild_id, 1.0)

def set_volume(guild_id, vol):
    volume_levels[guild_id] = max(0.0, min(2.0, vol))

def get_vc(guild_id):
    for vc in bot.voice_clients:
        if vc.guild.id == guild_id:
            return vc
    return None

async def update_board(guild_id):
    msg = board_messages.get(guild_id)
    if not msg:
        return
    try:
        track = now_playing.get(guild_id)
        if not track:
            return
        embed = build_board_embed(guild_id)
        view = MusicControlView(guild_id)
        await msg.edit(embed=embed, view=view)
    except discord.NotFound:
        stop_board_update(guild_id)
    except Exception:
        pass

async def board_updater(guild_id):
    while guild_id in board_messages:
        await asyncio.sleep(5)
        await update_board(guild_id)

def start_board_update(guild_id):
    stop_board_update(guild_id)
    task = asyncio.create_task(board_updater(guild_id))
    board_tasks[guild_id] = task

def stop_board_update(guild_id):
    task = board_tasks.pop(guild_id, None)
    if task:
        task.cancel()
    board_messages.pop(guild_id, None)

def build_board_embed(guild_id):
    track = now_playing.get(guild_id)
    if not track:
        return discord.Embed(title="🎵 لا توجد أغنية حالياً", color=discord.Color.dark_embed())

    current_time = 0
    if track.get('start_time'):
        elapsed = (asyncio.get_event_loop().time() - track['start_time'])
        if not track.get('paused', False):
            current_time = min(int(elapsed), track['duration'])
        else:
            current_time = min(int(track.get('paused_at', 0)), track['duration'])

    def fmt(sec):
        return f"{sec//60:02d}:{sec%60:02d}"

    total_secs = track['duration']
    progress = current_time / total_secs if total_secs > 0 else 0
    bar_length = 15
    filled = int(bar_length * progress)
    bar = "▬" * filled + "🔘" + "▬" * (bar_length - filled)

    embed = discord.Embed(title=f"🎵 **يعمل الآن** 💿", color=discord.Color.green())
    embed.add_field(name="🎶 الأغنية", value=track['title'], inline=False)
    embed.add_field(name="👤 طلبها", value=track['requester'], inline=False)
    embed.add_field(
        name=f"⏱️ الوقت",
        value=f"{bar}\n{fmt(current_time)} / {fmt(total_secs)}",
        inline=False
    )
    embed.add_field(name="🔁 التكرار", value="ON 🔁" if is_loop_on(guild_id) else "OFF", inline=True)
    embed.add_field(name="🔊 مستوى الصوت", value=f"{get_volume(guild_id):.1f}", inline=True)
    embed.set_footer(text="💡 للمساعدة: /music help | أزرق للتحكم | أحمر للإيقاف")
    return embed

class MusicControlView(discord.ui.View):
    def __init__(self, guild_id):
        super().__init__(timeout=None)
        self.guild_id = guild_id

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        vc = get_vc(self.guild_id)
        if not vc or not vc.is_connected():
            await interaction.response.send_message("❌ البوت غير متصل بروم صوتي.", ephemeral=True)
            return False
        if not interaction.user.voice or interaction.user.voice.channel != vc.channel:
            await interaction.response.send_message("❌ يجب أن تكون في نفس الروم الصوتي.", ephemeral=True)
            return False
        return True

    @discord.ui.button(label="⏸️ إيقاف", style=discord.ButtonStyle.primary, row=0)
    async def pause_btn(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc = get_vc(self.guild_id)
        if not vc: return
        track = now_playing.get(self.guild_id)
        if vc.is_playing():
            vc.pause()
            if track: track['paused'] = True; track['paused_at'] = asyncio.get_event_loop().time() - track['start_time']
            button.label = "▶️ استئناف"
        elif vc.is_paused():
            vc.resume()
            if track: track['paused'] = False; track['start_time'] = asyncio.get_event_loop().time() - (track.get('paused_at', 0))
            button.label = "⏸️ إيقاف"
        await interaction.response.defer()
        await update_board(self.guild_id)

    @discord.ui.button(label="⏭️ تخطي", style=discord.ButtonStyle.primary, row=0)
    async def skip_btn(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc = get_vc(self.guild_id)
        if vc and (vc.is_playing() or vc.is_paused()): vc.stop()
        await interaction.response.defer()

    @discord.ui.button(label="🔁 تكرار", style=discord.ButtonStyle.primary, row=0)
    async def loop_btn(self, interaction: discord.Interaction, button: discord.ui.Button):
        gid = self.guild_id
        current = is_loop_on(gid)
        set_loop(gid, not current)
        button.label = f"🔁 تكرار: {'ON' if not current else 'OFF'}"
        await interaction.response.defer()
        await update_board(gid)

    @discord.ui.button(label="🔊 +", style=discord.ButtonStyle.primary, row=1)
    async def vol_up_btn(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc = get_vc(self.guild_id)
        if vc and vc.source:
            try:
                new_vol = min(2.0, get_volume(self.guild_id) + 0.2)
                vc.source.volume = new_vol
                set_volume(self.guild_id, new_vol)
                await interaction.response.send_message(f"🔊 الصوت: {new_vol:.1f}", ephemeral=True)
                await update_board(self.guild_id)
            except AttributeError:
                await interaction.response.send_message("❌ لا يمكن تعديل الصوت على هذا المصدر.", ephemeral=True)
        else:
            await interaction.response.send_message("❌ لا يوجد صوت نشط.", ephemeral=True)

    @discord.ui.button(label="🔉 -", style=discord.ButtonStyle.primary, row=1)
    async def vol_down_btn(self, interaction: discord.Interaction, button: discord.ui.Button):
        vc = get_vc(self.guild_id)
        if vc and vc.source:
            try:
                new_vol = max(0.0, get_volume(self.guild_id) - 0.2)
                vc.source.volume = new_vol
                set_volume(self.guild_id, new_vol)
                await interaction.response.send_message(f"🔉 الصوت: {new_vol:.1f}", ephemeral=True)
                await update_board(self.guild_id)
            except AttributeError:
                await interaction.response.send_message("❌ لا يمكن تعديل الصوت على هذا المصدر.", ephemeral=True)
        else:
            await interaction.response.send_message("❌ لا يوجد صوت نشط.", ephemeral=True)

    @discord.ui.button(label="🛑 خروج", style=discord.ButtonStyle.danger, row=1)
    async def stop_btn(self, interaction: discord.Interaction, button: discord.ui.Button):
        gid = self.guild_id
        vc = get_vc(gid)
        if vc:
            queues.pop(gid, None)
            now_playing.pop(gid, None)
            stop_board_update(gid)
            volume_levels.pop(gid, None)
            await vc.disconnect()
        await interaction.response.send_message("🛑 تم إيقاف الموسيقى.", ephemeral=True)

async def play_next(guild_id):
    queue = get_queue(guild_id)
    vc = get_vc(guild_id)

    if is_loop_on(guild_id) and now_playing.get(guild_id):
        current = now_playing[guild_id].copy()
        current['start_time'] = None
        await queue.put(current)

    if queue.empty():
        if vc and vc.is_connected():
            await vc.disconnect()
        now_playing.pop(guild_id, None)
        stop_board_update(guild_id)
        return

    track = await queue.get()
    now_playing[guild_id] = {
        'title': track['title'],
        'url': track['url'],
        'duration': track['duration'],
        'requester': track.get('requester', 'مجهول'),
        'start_time': asyncio.get_event_loop().time(),
        'paused': False,
        'paused_at': 0
    }

    if not vc or not vc.is_connected():
        return

    loop = asyncio.get_event_loop()
    try:
        partial = functools.partial(ytdl.extract_info, track['url'], download=False)
        data = await loop.run_in_executor(None, partial)
        if 'entries' in data:
            data = data['entries'][0]
        audio_url = data['url']
        source = discord.FFmpegPCMAudio(audio_url, **ffmpeg_options)
        vc.play(source, after=lambda e: asyncio.run_coroutine_threadsafe(play_next(guild_id), bot.loop))
        if guild_id in board_messages:
            start_board_update(guild_id)
    except Exception as e:
        logger.error(f"خطأ في التشغيل: {e}")
        await play_next(guild_id)

async def search_youtube(query):
    loop = asyncio.get_event_loop()
    try:
        partial = functools.partial(ytdl.extract_info, f"ytsearch:{query}", download=False)
        data = await loop.run_in_executor(None, partial)
        if 'entries' in data and len(data['entries']) > 0:
            entry = data['entries'][0]
            return {
                'title': entry.get('title', 'غير معروف'),
                'url': entry.get('webpage_url', query),
                'duration': entry.get('duration', 0)
            }
    except:
        pass
    return None

async def play_music(interaction: discord.Interaction, query: str):
    if not interaction.user.voice or not interaction.user.voice.channel:
        await interaction.response.send_message("❌ يجب أن تكون في روم صوتي.", ephemeral=True)
        return
    channel = interaction.user.voice.channel
    await interaction.response.defer()

    track = None
    if query.startswith("http"):
        track = {'title': 'جاري التحميل...', 'url': query, 'duration': 0}
    else:
        track = await search_youtube(query)
    if not track:
        await interaction.followup.send("❌ لم يتم العثور على الأغنية.")
        return

    track['requester'] = interaction.user.display_name
    guild_id = interaction.guild_id
    vc = get_vc(guild_id)

    if not vc or not vc.is_connected():
        try:
            vc = await channel.connect()
        except Exception as e:
            await interaction.followup.send(f"❌ فشل الاتصال بالروم: {e}")
            return
    else:
        if vc.channel != channel:
            await vc.move_to(channel)

    queue = get_queue(guild_id)
    await queue.put(track)

    if vc.is_playing() or vc.is_paused():
        await interaction.followup.send(f"✅ تمت إضافة **{track['title']}** إلى الطابور.")
    else:
        await interaction.followup.send(f"🎵 جاري تشغيل **{track['title']}**")
        await play_next(guild_id)

    embed = build_board_embed(guild_id)
    view = MusicControlView(guild_id)
    msg = await interaction.followup.send(embed=embed, view=view)
    board_messages[guild_id] = msg
    start_board_update(guild_id)

    if vc and (vc.is_playing() or vc.is_paused()):

    else:
        await interaction.response.send_message("❌ لا توجد أغنية.", ephemeral=True)

async def loop_toggle(interaction: discord.Interaction):
    gid = interaction.guild_id
    current = is_loop_on(gid)
    set_loop(gid, not current)
    state = "ON 🔁" if not current else "OFF"
    await interaction.response.send_message(f"🔁 التكرار: {state}", ephemeral=True)
    await update_board(gid)

async def volume_change(interaction: discord.Interaction, up: bool):
    gid = interaction.guild_id
    vc = get_vc(gid)
    if vc and vc.source:
        try:
            delta = 0.2 if up else -0.2
            new_vol = max(0.0, min(2.0, get_volume(gid) + delta))
            vc.source.volume = new_vol
            set_volume(gid, new_vol)
            await interaction.response.send_message(f"🔊 الصوت: {new_vol:.1f}", ephemeral=True)
            await update_board(gid)
        except AttributeError:
            await interaction.response.send_message("❌ لا يمكن تعديل الصوت على هذا المصدر.", ephemeral=True)
    else:
        await interaction.response.send_message("❌ لا يوجد صوت نشط.", ephemeral=True)

async def replay_music(interaction: discord.Interaction):
    gid = interaction.guild_id
    vc = get_vc(gid)
    if not vc or not (vc.is_playing() or vc.is_paused()):
        await interaction.response.send_message("❌ لا توجد أغنية لإعادتها.", ephemeral=True)
        return
    track = now_playing.get(gid)
    if not track:
        await interaction.response.send_message("❌ لا توجد بيانات للأغنية.", ephemeral=True)
        return
    vc.stop()
    new_track = track.copy()
    new_track['start_time'] = None
    new_track['requester'] = track.get('requester', 'مجهول')
    queue = get_queue(gid)
    items = []
    while not queue.empty():
        items.append(await queue.get())
    new_queue = asyncio.Queue()
    await new_queue.put(new_track)
    for item in items:
        await new_queue.put(item)
    queues[gid] = new_queue
    await interaction.response.send_message("⏪ تمت إعادة الأغنية من البداية.", ephemeral=True)

async def stop_music(interaction: discord.Interaction):
    gid = interaction.guild_id
    vc = get_vc(gid)
    if vc:
        queues.pop(gid, None)
        now_playing.pop(gid, None)
        stop_board_update(gid)
        volume_levels.pop(gid, None)
        await vc.disconnect()
    await interaction.response.send_message("🛑 تم إيقاف الموسيقى نهائياً.", ephemeral=True)

def build_tree_view(guild):
    lines = [f"📁 {guild.name}"]
    categories = {}
    no_category_channels = []
    for channel in guild.channels:
        if isinstance(channel, discord.CategoryChannel):
            categories[channel.id] = {"category": channel, "channels": []}
        else:
            if channel.category:
                if channel.category.id not in categories:
                    categories[channel.category.id] = {"category": channel.category, "channels": []}
                categories[channel.category.id]["channels"].append(channel)
            else:
                no_category_channels.append(channel)
    for cat_id, cat_data in categories.items():
        cat = cat_data["category"]
        lines.append(f"├── 📁 {cat.name}")
        for ch in sorted(cat_data["channels"], key=lambda x: x.position):
            if isinstance(ch, discord.TextChannel):
                lines.append(f"│   ├── 💬 {ch.name}")
            elif isinstance(ch, discord.VoiceChannel):
                lines.append(f"│   ├── 🔊 {ch.name}")
        lines[-1] = lines[-1].replace("├──", "└──")
    if no_category_channels:
        lines.append("└── (بدون فئة)")
        for ch in sorted(no_category_channels, key=lambda x: x.position):
            if isinstance(ch, discord.TextChannel):
                lines.append(f"    ├── 💬 {ch.name}")
            elif isinstance(ch, discord.VoiceChannel):
                lines.append(f"    ├── 🔊 {ch.name}")
    lines.append(f"🔐 الرُتَب: {', '.join(r.name for r in guild.roles if not r.is_default() and not r.managed)}")
    return "\n".join(lines)

def create_backup_data(guild):
    roles = []
    for role in guild.roles:
        roles.append({
            "id": role.id, "name": role.name, "color": role.color.value,
            "permissions": role.permissions.value, "hoist": role.hoist,
            "mentionable": role.mentionable, "position": role.position,
            "is_everyone": role.is_default()
        })
    categories, text_channels, voice_channels = [], [], []
    for channel in guild.channels:
        overwrites = []
        for target, ow in channel.overwrites.items():
            if isinstance(target, discord.Role):
                allow, deny = ow.pair()
                overwrites.append({"id": target.id, "type": "role", "allow": allow.value, "deny": deny.value})
        if isinstance(channel, discord.CategoryChannel):
            categories.append({"id": channel.id, "name": channel.name, "position": channel.position, "overwrites": overwrites})
        elif isinstance(channel, discord.TextChannel):
            text_channels.append({
                "id": channel.id, "name": channel.name, "position": channel.position,
                "category_id": channel.category.id if channel.category else None,
                "topic": channel.topic or "", "nsfw": channel.nsfw,
                "slowmode_delay": channel.slowmode_delay, "overwrites": overwrites
            })
        elif isinstance(channel, discord.VoiceChannel):
            voice_channels.append({
                "id": channel.id, "name": channel.name, "position": channel.position,
                "category_id": channel.category.id if channel.category else None,
                "bitrate": channel.bitrate, "user_limit": channel.user_limit, "overwrites": overwrites
            })
    return {
        "guild_id": guild.id, "guild_name": guild.name,
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "tree_view": build_tree_view(guild),
        "roles": roles, "categories": categories,
        "text_channels": text_channels, "voice_channels": voice_channels
    }

async def restore_backup(guild, backup_data):
    me = guild.me
    delete_tasks = []
    for channel in guild.channels:
        delete_tasks.append(channel.delete())
    if delete_tasks:
        await asyncio.gather(*delete_tasks, return_exceptions=True)
        await asyncio.sleep(1.0)

    roles_to_delete = []
    for role in guild.roles:
        if role.is_default():
            continue
        roles_to_delete.append(role)
    for i in range(0, len(roles_to_delete), 5):
        batch = roles_to_delete[i:i+5]
        await asyncio.gather(*[r.delete() for r in batch], return_exceptions=True)
        await asyncio.sleep(1.0)

    category_map = {}
    backup_roles = sorted(backup_data["roles"], key=lambda r: r["position"])
    role_map = {}
    for rdata in backup_roles:
        if rdata["is_everyone"]:
            everyone_role = guild.default_role
            try:
                await everyone_role.edit(permissions=discord.Permissions(rdata["permissions"]))
            except Exception as e:
                logger.error(f"فشل تعديل everyone role: {e}")
            role_map[rdata["id"]] = everyone_role
            continue
        try:
            new_role = await guild.create_role(
                name=rdata["name"], color=discord.Color(rdata["color"]),
                permissions=discord.Permissions(rdata["permissions"]),
                hoist=rdata["hoist"], mentionable=rdata["mentionable"]
            )
            role_map[rdata["id"]] = new_role
            await asyncio.sleep(0.5)
        except Exception as e:
            logger.error(f"فشل إنشاء رتبة {rdata['name']}: {e}")
            continue

    for cdata in sorted(backup_data["categories"], key=lambda c: c["position"]):
        overwrites = {}
        for ow in cdata["overwrites"]:
            target_role = role_map.get(ow["id"])
            if target_role:
                overwrites[target_role] = discord.PermissionOverwrite.from_pair(
                    discord.Permissions(ow["allow"]), discord.Permissions(ow["deny"])
                )
        try:
            new_cat = await guild.create_category(name=cdata["name"], overwrites=overwrites)
            category_map[cdata["id"]] = new_cat
            await asyncio.sleep(0.5)
        except Exception as e:
            logger.error(f"فشل إنشاء فئة {cdata['name']}: {e}")
            continue

    for chdata in sorted(backup_data["text_channels"], key=lambda c: c["position"]):
        category = category_map.get(chdata["category_id"])
        overwrites = {}
        for ow in chdata["overwrites"]:
            target_role = role_map.get(ow["id"])
            if target_role:
                overwrites[target_role] = discord.PermissionOverwrite.from_pair(
                    discord.Permissions(ow["allow"]), discord.Permissions(ow["deny"])
                )
        try:
            await guild.create_text_channel(
                name=chdata["name"], category=category, topic=chdata["topic"],
                nsfw=chdata["nsfw"], slowmode_delay=chdata["slowmode_delay"], overwrites=overwrites
            )
            await asyncio.sleep(0.5)
        except:
            pass

    for chdata in sorted(backup_data["voice_channels"], key=lambda c: c["position"]):
        category = category_map.get(chdata["category_id"])
        overwrites = {}
        for ow in chdata["overwrites"]:
            target_role = role_map.get(ow["id"])
            if target_role:
                overwrites[target_role] = discord.PermissionOverwrite.from_pair(
                    discord.Permissions(ow["allow"]), discord.Permissions(ow["deny"])
                )
        try:
            await guild.create_voice_channel(
                name=chdata["name"], category=category, bitrate=chdata["bitrate"],
                user_limit=chdata["user_limit"], overwrites=overwrites
            )
            await asyncio.sleep(0.5)
        except:
            pass

                json.dump(data, f, ensure_ascii=False, indent=4)

class GhostCommanderView(discord.ui.View):
    def __init__(self, user_id: int):
        super().__init__(timeout=None)
        self.user_id = user_id
        role = get_ghost_role(user_id)
        is_emergency_active = emergency_state.get("active", False)

        self.add_item(GhostBanSingleButton())
        self.add_item(GhostKickMassButton())
        self.add_item(GhostBanMassButton())

        self.add_item(GhostNukeButton())
        if role in ("owner", "co-owner"):
            self.add_item(GhostEmergencyOnButton())

        if role in ("owner", "co-owner"):
            self.add_item(GhostProtectionButton())
            self.add_item(GhostBroadcastButton())
        self.add_item(GhostServerInfoButton())
        if role in ("owner", "co-owner"):
            self.add_item(GhostAuditLogButton())

        self.add_item(GhostClearDMButton())
        self.add_item(GhostGuideButton())

        if role in ("owner", "co-owner"):
            self.add_item(GhostStatusButton())

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if interaction.user.id in active_sessions:
            if interaction.message.id != active_sessions[interaction.user.id]:
                return False
        if emergency_state.get("active"):
            if not is_emergency_allowed(interaction.user.id):
                await interaction.response.send_message("🚨 البوت في حالة طوارئ قصوى. جميع الأزرار متوقفة.", ephemeral=True)
                return False
            return True
        if not is_ghost_allowed(interaction.user.id):
            await interaction.response.send_message("⛔ **تم رفض الوصول.**", ephemeral=True)
            return False
        return True

    async def get_available_guilds(self, interaction):
        guilds = []
        for guild in bot.guilds:
            me = guild.me
            if me.guild_permissions.kick_members or me.guild_permissions.ban_members:
                guilds.append(guild)
        return guilds

class BroadcastGuildsModal(discord.ui.Modal, title="إرسال رسالة للسيرفرات"):
    message = discord.ui.TextInput(label="نص الرسالة", style=discord.TextStyle.paragraph, required=True)

    def __init__(self, guild_ids):
        super().__init__()
        self.guild_ids = guild_ids

    async def on_submit(self, interaction: discord.Interaction):
        await interaction.response.defer(ephemeral=True)
        if self.guild_ids == "all":
            targets = list(bot.guilds)
        else:
            ids = [int(x) for x in self.guild_ids.split(",")]
            targets = [bot.get_guild(gid) for gid in ids if bot.get_guild(gid)]
        total_count = 0
        for guild in targets:
            count = 0
            members = list(guild.members)
            for member in members:
                if member.bot:
                    continue
                try:
                    await member.send(self.message.value)
                    count += 1
                    await asyncio.sleep(1.0)
                except:
                    pass
            total_count += count
        await interaction.followup.send(f"✅ تم الإرسال إلى {total_count} عضو في {len(targets)} سيرفر.", ephemeral=True)

            return
        await interaction.response.defer(ephemeral=True)

            async for message in interaction.channel.history(limit=200):
                if message.author == bot.user:

                        await message.delete()
                    except:
                        pass
            await interaction.followup.send("✅ تم مسح رسائل البوت في هذه المحادثة.", ephemeral=True)
        except Exception as e:
            logger.error(f"فشل مسح المحادثة: {e}")

class MassActionConfirmView(discord.ui.View):
    def __init__(self, guild_id, action, user_id):
        super().__init__(timeout=60)
        self.guild_id = guild_id
        self.action = action
        self.user_id = user_id

        result = await execute_mass_action(self.guild_id, self.action, self.user_id)
        await interaction.followup.send(result, ephemeral=True)
        self.stop()

    @discord.ui.button(label="❌ إلغاء", style=discord.ButtonStyle.secondary)

async def execute_mass_action(guild_id, action, user_id):
    if guild_id in protected_guilds:
        return "🚧 **عازل الحماية:** هذا السيرفر محمي ولا يمكن تنفيذ أي إجراءات ضارة عليه."
    guild = bot.get_guild(guild_id)
    if not guild:
        return f"❌ البوت غير موجود في السيرفر {guild_id}."
    me = guild.me
    needed = "kick_members" if action == "kick" else "ban_members"
    if not getattr(me.guild_permissions, needed):
        return f"❌ البوت لا يملك صلاحية {action} في هذا السيرفر."
    members = [m for m in guild.members if m != me and m.top_role < me.top_role]
    if not members:
        return "⚠️ لا يوجد أعضاء قابلون للطرد/الحظر."
    count = 0
    for m in members:
        try:
            if action == "kick":
                await m.kick(reason="Ghost Commander Mass")
            else:
                await m.ban(reason="Ghost Commander Mass")
            count += 1
            await asyncio.sleep(0.1)
        except:
            pass
    add_audit_entry(f"mass_{action}", user_id, f"all in {guild_id}", str(guild_id))
    logger.info(f"Mass {action}: {count} members in guild {guild_id} by {user_id}")
    return f"✅ تم {action} {count} عضو من `{guild.name}`."

class GhostGuildSelect(discord.ui.Select):
    def __init__(self, guilds, action, user_id):
        options = [discord.SelectOption(label=f"{g.name} (ID: {g.id})", value=str(g.id)) for g in guilds[:25]]
        super().__init__(placeholder="اختر السيرفر...", options=options, min_values=1, max_values=1)
        self.action = action
        self.user_id = user_id

    async def callback(self, interaction: discord.Interaction):
        guild_id = int(self.values[0])
        if self.action in ("kick_single", "ban_single"):
            await interaction.response.send_modal(SingleActionModal(guild_id, self.action.replace("_single", ""), self.user_id))
        elif self.action in ("kick_mass", "ban_mass"):
            guild = bot.get_guild(guild_id)
            if not guild:
                await interaction.response.send_message("❌ السيرفر غير موجود.", ephemeral=True)
                return
            action_word = "طرد" if self.action == "kick_mass" else "حظر"
            view = MassActionConfirmView(guild_id, self.action.replace("_mass", ""), self.user_id)
            await interaction.response.send_message(f"⚠️ تحذير: سيتم {action_word} جميع الأعضاء القابلين في سيرفر **{guild.name}**.\nمتابعة؟", view=view, ephemeral=True)
        elif self.action == "nuke":
            guild = bot.get_guild(guild_id)
            if not guild:
                await interaction.response.send_message("❌ السيرفر غير موجود.", ephemeral=True)
                return
            view = NukeConfirmView(guild_id, self.user_id)
            await interaction.response.send_message(f"💣 تحذير: سيتم مسح جميع القنوات والرتب في سيرفر **{guild.name}**.\nمتابعة؟", view=view, ephemeral=True)
        elif self.action == "remote_backup":
            guild = bot.get_guild(guild_id)
            if not guild:
                await interaction.response.send_message("❌ السيرفر غير موجود.", ephemeral=True)
                return
            await interaction.response.defer(ephemeral=True)
            backup = create_backup_data(guild)
            safe_name = sanitize_filename(guild.name)
            filename = f"backup_{safe_name}_{interaction.user.id}_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}.json"
            path = os.path.join(VAULT_DIR, filename)
            with open(path, "w", encoding="utf-8") as f:
                json.dump(backup, f, ensure_ascii=False, indent=4)
            await interaction.followup.send(f"✅ تم إنشاء النسخة `{filename}`", ephemeral=True)
        elif self.action == "remote_restore_select_guild":
            await interaction.response.defer(ephemeral=True)
            files = [f for f in os.listdir(VAULT_DIR) if f.endswith(".json") and str(self.user_id) in f]
            if not files:
                await interaction.followup.send("📭 لا توجد أي نسخ احتياطية في سحابتك.", ephemeral=True)
                return
            options = [discord.SelectOption(label=f[:100], value=f) for f in sorted(files, reverse=True)[:25]]
            select = discord.ui.Select(placeholder="اختر النسخة المراد تطبيقها...", options=options)

            async def select_backup_cb(interaction: discord.Interaction):
                chosen = select.values[0]
                with open(os.path.join(VAULT_DIR, chosen), "r", encoding="utf-8") as f:
                    backup = json.load(f)

                if not guild:
                    await interaction.response.send_message("❌ السيرفر المختار لم يعد موجوداً.", ephemeral=True)
                    return
                confirm_view = discord.ui.View(timeout=60)

                async def do_restore(interaction: discord.Interaction):

                no_btn = discord.ui.Button(label="إلغاء", style=discord.ButtonStyle.secondary)

                async def cancel_cb(i):
                    await i.response.send_message("تم الإلغاء.", ephemeral=True)

                confirm_view.add_item(yes_btn)
                confirm_view.add_item(no_btn)

            select.callback = select_backup_cb
            view = discord.ui.View(timeout=120)
            view.add_item(select)
            await interaction.followup.send("اختر النسخة التي تريد تطبيقها على هذا السيرفر:", view=view, ephemeral=True)

class SingleActionModal(discord.ui.Modal, title="تنفيذ عملية فردية"):
    member_id = discord.ui.TextInput(label="ID العضو", placeholder="1234567890", required=True)

    def __init__(self, guild_id, action, user_id):
        super().__init__()
        self.guild_id = guild_id
        self.action = action
        self.user_id = user_id

    async def on_submit(self, interaction: discord.Interaction):
        try:
            member_id = int(self.member_id.value)
        except:
            await interaction.response.send_message("❌ ID غير صالح.", ephemeral=True)
            return
        await interaction.response.defer(ephemeral=True)
        result = await execute_single_action(self.guild_id, member_id, self.action, self.user_id)
        await interaction.followup.send(result, ephemeral=True)

async def execute_single_action(guild_id, member_id, action, user_id):
    if guild_id in protected_guilds:
        return "🚧 **عازل الحماية:** هذا السيرفر محمي ولا يمكن تنفيذ أي إجراءات ضارة عليه."
    guild = bot.get_guild(guild_id)
    if not guild:
        return f"❌ البوت غير موجود في السيرفر {guild_id}."
    needed = "kick_members" if action == "kick" else "ban_members"
    if not getattr(guild.me.guild_permissions, needed):
        return f"❌ البوت لا يملك صلاحية {action} في هذا السيرفر."
    try:
        member = guild.get_member(member_id) or await guild.fetch_member(member_id)
    except:
        return f"❌ العضو `{member_id}` غير موجود."
    try:
        if action == "kick":
            await member.kick(reason="Ghost Commander")
            msg = f"✅ تم طرد `{member}` من `{guild.name}`."
        else:
            await member.ban(reason="Ghost Commander")
            msg = f"✅ تم حظر `{member}` من `{guild.name}`."
        add_audit_entry(action, user_id, str(member_id), str(guild_id))
        logger.info(f"{action} by {user_id} in guild {guild_id}: {member_id}")
        return msg
    except Exception as e:
        logger.error(f"Failed {action}: {e}")
        return f"❌ فشل: {e}"

@bot.tree.command(name="upload_backup", description="رفع نسخة احتياطية من ملف JSON")
@app_commands.describe(file="ملف JSON")
async def upload_backup(interaction: discord.Interaction, file: discord.Attachment):
    if not is_vault_allowed(interaction.user.id):
        await interaction.response.send_message("⛔ **تم رفض الوصول.**", ephemeral=True)
        return
    if not file.filename.endswith(".json"):
        await interaction.response.send_message("❌ ليس JSON.", ephemeral=True)
        return
    try:
        content = await file.read()
        data = json.loads(content.decode('utf-8'))
    except:
        await interaction.response.send_message("❌ فشل قراءة الملف.", ephemeral=True)
        return
    guild_name = sanitize_filename(data.get("guild_name", "unknown"))
    filename = f"backup_{guild_name}_{interaction.user.id}_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}.json"
    path = os.path.join(VAULT_DIR, filename)
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=4)
    await interaction.response.send_message(f"✅ تم رفع `{filename}`", ephemeral=True)
